function createSongCard(song, index) {
    const card = document.createElement('div');
    card.className = 'card';
    card.innerHTML = `
        <div class="cover" style="background: ${song.gradient}">
            <span class="badge">Trending</span>
            <button class="play-btn" aria-label="Play ${song.title}">▶️</button>
        </div>
        <div class="card-info">
            <h4>${song.title}</h4>
            <p>${song.artist}</p>
        </div>
    `;
    card.querySelector('.play-btn').addEventListener('click', (e) => {
        e.stopPropagation();
        playSongAt(index);
    });
    return card;
}

// PLAYER LOGIC
// ============================================================
function playSongAt(index) {
    currentIndex = index;
    const song = songs[currentIndex];
    audio.src = song.src;
    audio.volume = 0.7;
    // Update CSS cover
    document.getElementById('player-cover').style.background = song.gradient;
    document.getElementById('player-title').textContent = song.title;
    document.getElementById('player-artist').textContent = song.artist;
    updateFavoriteBtn();
    audio.play();
    updatePlayBtn();
}

function togglePlay() {
    if (audio.paused) {
        if (!audio.src) loadCurrentSong();
        audio.play();
    } else {
        audio.pause();
    }
}

function loadCurrentSong() {
    const song = songs[currentIndex];
    audio.src = song.src;
    document.getElementById('player-cover').style.background = song.gradient;
    document.getElementById('player-title').textContent = song.title;
    document.getElementById('player-artist').textContent = song.artist;
}

function nextSong() {
    let index;
    if (isShuffle) {
        index = Math.floor(Math.random() * songs.length);
        if (index === currentIndex) index = (index + 1) % songs.length;
    } else {
        index = (currentIndex + 1) % songs.length;
    }
    playSongAt(index);
}

function prevSong() {
    const index = (currentIndex - 1 + songs.length) % songs.length;
    playSongAt(index);
}

function updatePlayButton() {
    isPlaying = !audio.paused;
    playBtn.textContent = isPlaying ? '⏸️' : '▶️';
}

function updateProgress() {
    if (audio.duration) {
        const percent = (audio.currentTime / audio.duration) * 100;
        progressFill.style.width = percent + '%';
        currentTimeEl.textContent = formatTime(audio.currentTime);
        durationEl.textContent = formatTime(audio.duration);
    }
}

function formatTime(seconds) {
    const mins = Math.floor(seconds / 60);
    const secs = Math.floor(seconds % 60);
    return `${mins}:${secs.toString().padStart(2, '0')}`;
}

function toggleFavorite() {
    const song = songs[currentIndex];
    if (favorites.has(song.title)) {
        favorites.delete(song.title);
        favoriteBtn.classList.remove('active');
    } else {
        favorites.add(song.title);
        favoriteBtn.classList.add('active');
    }
}

function updateFavoriteBtn() {
    favoriteBtn.classList.toggle('active', favorites.has(songs[currentIndex]?.title));
}
